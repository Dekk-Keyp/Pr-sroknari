/** * Dekk-Keyp — live WooCommerce feed for prísroknarin. * * The catalogue is too big to crawl while a page waits, so the crawl never * happens on the critical path: a scheduled job keeps a copy in KV, and a * request is answered from that copy immediately. If the copy is getting old * the refresh runs in the background, after the response has already gone out. * * Bindings * FEED KV namespace (Settings → Bindings → KV namespace, name it FEED) * Secrets * WC_KEY consumer key (read-only) * WC_SECRET consumer secret * Text vars * WC_SITE https://dekk-keyp.fo * ORIGINS comma-separated list of allowed origins */ const COLS = [ 'ID', 'Type', 'Name', 'Stock', 'Regular price', 'Sale price', 'Product URL', 'SKU', 'No of Sales', 'Attributes', 'Categories', 'Tags', 'Status', 'Fjarlagur', 'Tøkt íalt', 'Goymsluprísur' ]; /** Only the fields the CSV needs — this cuts the payload by roughly 90%. */ const FIELDS = [ 'id', 'type', 'name', 'manage_stock', 'stock_quantity', 'stock_status', 'regular_price', 'sale_price', 'permalink', 'sku', 'total_sales', 'attributes', 'categories', 'tags', 'meta_data' ].join(','); const KV_KEY = 'catalogue'; const PER_PAGE = 100; const MAX_PAGES = 60; const PAGE_PAUSE_MS = 500; // stay under Cloudflare's rate limit const MAX_RETRIES = 4; const REFRESH_MS = 10 * 60 * 1000; // older than this: refresh in background const RETRY_STATUSES = [429, 500, 502, 503, 504]; const sleep = ms => new Promise(r => setTimeout(r, ms)); function csvCell(v) { const s = v === null || v === undefined ? '' : String(v); return /[",\n\r]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s; } function csvRow(arr) { return arr.map(csvCell).join(','); } /** "pa_breidd: 235, pa_profil: 50, pa_arstid: vetrardekk" */ const WANTED_ATTRS = ['pa_breidd', 'pa_profil', 'pa_tvormal', 'pa_arstid', 'pa_load-index', 'pa_speed-index']; function attrString(product) { const out = []; for (const a of product.attributes || []) { const slug = a.slug || (a.name ? 'pa_' + String(a.name).toLowerCase().replace(/\s+/g, '-') : ''); if (!slug || WANTED_ATTRS.indexOf(slug) === -1) continue; const val = Array.isArray(a.options) ? a.options.join('|') : (a.option || ''); if (val === '') continue; out.push(slug + ': ' + val); } return out.join(', '); } /** Find a meta_data value whose key contains `needle` (case-insensitive). */ function meta(product, needle) { const n = needle.toLowerCase(); for (const m of product.meta_data || []) { const k = String(m.key || '').toLowerCase(); if (k.includes(n)) { const v = m.value; if (v === null || v === undefined) return ''; return typeof v === 'object' ? JSON.stringify(v) : String(v); } } return ''; } /** * Stock column: a number when WooCommerce manages stock, otherwise the status * string. The calculator reads this column both as a quantity and for the text * "outofstock", which is how the sheet behaves too. */ function stockCell(p) { if (p.manage_stock && p.stock_quantity !== null && p.stock_quantity !== undefined) { return p.stock_quantity; } return p.stock_status || ''; } function productRow(p) { return [ p.id, p.type, p.name, stockCell(p), p.regular_price, p.sale_price, p.permalink, p.sku, p.total_sales, attrString(p), (p.categories || []).map(c => c.name).join('|'), (p.tags || []).map(t => t.name).join('|'), p.stock_status, meta(p, 'fjarlag'), meta(p, 'tokt') || meta(p, 'tøkt'), meta(p, 'goymslu') ]; } /** Name who refused us, so a 429 is diagnosable instead of mysterious. */ function describeRefusal(res, body, page) { const interesting = ['retry-after', 'server', 'x-powered-by', 'cf-ray', 'x-litespeed-cache', 'x-ratelimit-limit', 'x-ratelimit-remaining', 'x-ratelimit-reset']; const seen = []; for (const name of interesting) { const v = res.headers.get(name); if (v) seen.push(name + '=' + v); } const snippet = String(body || '').replace(/\s+/g, ' ').trim().slice(0, 200); return 'WooCommerce ' + res.status + ' á síðu ' + page + (seen.length ? ' [' + seen.join(', ') + ']' : '') + (snippet ? ' — ' + snippet : ''); } /** Header values must be ASCII — Faroese text in one throws and kills the response. */ function headerSafe(s) { return String(s).replace(/[^\x20-\x7E]/g, '?').slice(0, 200); } async function fetchPage(url, auth, page) { for (let attempt = 0; ; attempt++) { const res = await fetch(url, { headers: { Authorization: auth, Accept: 'application/json', 'User-Agent': 'Dekk-Keyp-Prisroknari/1.0 (Cloudflare Worker)' } }); if (res.ok) return res; const retryable = RETRY_STATUSES.indexOf(res.status) > -1; if (!retryable || attempt >= MAX_RETRIES) { throw new Error(describeRefusal(res, await res.text(), page)); } const ra = parseInt(res.headers.get('Retry-After') || '', 10); await sleep(Number.isFinite(ra) && ra > 0 ? Math.min(ra * 1000, 10000) : Math.min(1000 * Math.pow(2, attempt), 10000)); } } /** Crawl the catalogue. A run cut short still returns what it managed to read. */ async function crawl(env) { const auth = 'Basic ' + btoa(env.WC_KEY + ':' + env.WC_SECRET); const site = String(env.WC_SITE || '').replace(/\/+$/, ''); const rows = []; let page = 1; while (page <= MAX_PAGES) { const url = site + '/wp-json/wc/v3/products' + '?per_page=' + PER_PAGE + '&page=' + page + '&status=publish&orderby=id&order=asc' + '&_fields=' + encodeURIComponent(FIELDS); let batch; try { batch = await (await fetchPage(url, auth, page)).json(); } catch (err) { return { rows, complete: false, error: err.message }; } if (!Array.isArray(batch) || batch.length === 0) break; for (const p of batch) rows.push(productRow(p)); if (batch.length < PER_PAGE) break; page++; await sleep(PAGE_PAUSE_MS); } return { rows, complete: true, error: null }; } /** Crawl and store. Returns the stored record, or null if nothing was read. */ async function refresh(env) { const { rows, complete, error } = await crawl(env); if (rows.length === 0) return null; const csv = [csvRow(COLS)].concat(rows.map(csvRow)).join('\n'); const rec = { csv, at: Date.now(), rows: rows.length, complete, error }; await putStore(env, rec); return rec; } // ---- storage: KV when bound, Cache API otherwise ------------------------- async function getStore(env, origin) { if (env.FEED) { const got = await env.FEED.getWithMetadata(KV_KEY, { type: 'text' }); if (got && got.value) { const m = got.metadata || {}; return { csv: got.value, at: m.at || 0, rows: m.rows || 0, complete: m.complete !== false, error: m.error || null }; } return null; } const hit = await caches.default.match(new Request(origin + '/__feed')); if (!hit) return null; return { csv: await hit.text(), at: parseInt(hit.headers.get('X-Dekk-At') || '0', 10), rows: parseInt(hit.headers.get('X-Dekk-Rows') || '0', 10), complete: hit.headers.get('X-Dekk-Complete') !== 'no', error: hit.headers.get('X-Dekk-Error') || null }; } async function putStore(env, rec) { if (env.FEED) { await env.FEED.put(KV_KEY, rec.csv, { metadata: { at: rec.at, rows: rec.rows, complete: rec.complete, error: rec.error ? headerSafe(rec.error) : null } }); return; } if (!env.__origin) return; await caches.default.put(new Request(env.__origin + '/__feed'), new Response(rec.csv, { headers: { 'Cache-Control': 'public, max-age=86400', 'X-Dekk-At': String(rec.at), 'X-Dekk-Rows': String(rec.rows), 'X-Dekk-Complete': rec.complete ? 'yes' : 'no', 'X-Dekk-Error': rec.error ? headerSafe(rec.error) : '' } })); } // ---- http ---------------------------------------------------------------- function corsFor(request, env) { const origin = request.headers.get('Origin') || ''; const allowed = String(env.ORIGINS || '').split(',').map(s => s.trim()).filter(Boolean); const headers = { 'Vary': 'Origin' }; if (origin && allowed.indexOf(origin) > -1) headers['Access-Control-Allow-Origin'] = origin; else if (allowed.indexOf('*') > -1) headers['Access-Control-Allow-Origin'] = '*'; headers['Access-Control-Allow-Methods'] = 'GET, OPTIONS'; return headers; } function csvResponse(rec, cors, state) { const h = new Headers({ 'Content-Type': 'text/plain; charset=utf-8', 'Cache-Control': 'no-store', 'X-Dekk-Rows': String(rec.rows), 'X-Dekk-Complete': rec.complete ? 'yes' : 'no', 'X-Dekk-Age-Seconds': String(Math.round((Date.now() - rec.at) / 1000)), 'X-Dekk-State': state }); if (rec.error) h.set('X-Dekk-Error', headerSafe(rec.error)); for (const k in cors) h.set(k, cors[k]); return new Response(rec.csv, { status: 200, headers: h }); } export default { async scheduled(event, env, ctx) { ctx.waitUntil(refresh(env)); }, async fetch(request, env, ctx) { const cors = corsFor(request, env); if (request.method === 'OPTIONS') return new Response(null, { status: 204, headers: cors }); if (request.method !== 'GET') return new Response('Method not allowed', { status: 405, headers: cors }); const origin = new URL(request.url).origin; env.__origin = origin; // 1. answer from the stored copy straight away const rec = await getStore(env, origin); if (rec && rec.csv) { const stale = (Date.now() - rec.at) > REFRESH_MS; // refresh AFTER responding, so nobody waits for the crawl if (stale) ctx.waitUntil(refresh(env).catch(() => {})); return csvResponse(rec, cors, stale ? 'stale-refreshing' : 'fresh'); } // 2. nothing stored yet. Start the crawl, but do NOT make this page wait // for it: answer at once so the calculator falls through to its other // source. By the next load the copy is in place and replies are instant. ctx.waitUntil(refresh(env).catch(() => {})); return new Response('FEILUR: vorulistin verdur heintadur - royn aftur um eitt minutt', { status: 503, headers: new Headers(Object.assign({ 'Content-Type': 'text/plain; charset=utf-8', 'Cache-Control': 'no-store', 'Retry-After': '60', 'X-Dekk-State': 'warming' }, cors)) }); } };
